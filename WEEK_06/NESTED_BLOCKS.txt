---   JOINS 
DECLARE 

F_NAME VARCHAR2(20);
LAST_NAME VARCHAR2(10);
SAL NUMBER(10);
CITY VARCHAR2(20);
DEPARTMENT_NAME VARCHAR2(20);


BEGIN 
     SELECT E.FIRST_NAME , E.LAST_NAME , E.SALARY  , S.CITY , D.DEPARTMENT_NAME  INTO  F_NAME , LAST_NAME , SAL , CITY , DEPARTMENT_NAME  
	 FROM EMPLOYEES E  JOIN DEPARTMENTS D
	 ON (E.DEPARTMENT_ID = D.DEPARTMENT_ID)
	 JOIN LOCATIONS S 
	 ON(D.LOCATION_ID = S.LOCATION_ID)
	 WHERE E.DEPARTMENT_ID = 10;
	 
	 DBMS_OUTPUT.PUT_LINE (F_NAME ||' '|| LAST_NAME ||' '||  SAL ||' '|| CITY ||' '|| DEPARTMENT_NAME );
	 
END ;

SET SERVEROUTPUT ON 
DECLARE
F_NAME VARCHAR2(20);
L_NAME VARCHAR2(20);
SALARY NUMBER(10);

BEGIN 

SELECT FIRST_NAME , LAST_NAME , SALARY  INTO F_NAME , L_NAME , SALARY 
FROM EMPLOYEES
WHERE SALARY = (SELECT MAX(SALARY)
FROM EMPLOYEES );

DBMS_OUTPUT.PUT_LINE(F_NAME ||' '|| L_NAME||' '||SALARY);


END ;

SET SERVEROUTPUT ON;

DECLARE 
    v_first_name  VARCHAR2(20);
    v_sal         NUMBER(10);
    v_new_com     NUMBER(10);
BEGIN 

    SELECT FIRST_NAME,
           SALARY,
           CASE 
               WHEN COMMISSION_PCT IS NULL THEN 0
               ELSE COMMISSION_PCT
           END
    INTO   v_first_name,
           v_sal,
           v_new_com
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID = 10
   

    DBMS_OUTPUT.PUT_LINE(
        v_first_name || ' ' || v_sal || ' ' || v_new_com
    );

END;
