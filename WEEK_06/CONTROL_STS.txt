DECLARE 

F_NAME VARCHAR2(20);
LAST_NAME VARCHAR2(10);
SAL NUMBER(10);
CITY VARCHAR2(20);
DEPARTMENT_NAME VARCHAR2(20);


BEGIN 
     SELECT E.FIRST_NAME , E.LAST_NAME , E.SALARY  , S.CITY , D.DEPARTMENT_NAME  INTO  F_NAME , LAST_NAME , SAL , CITY , DEPARTMENT_NAME  
	 FROM EMPLOYEES E  JOIN DEPARTMENTS D
	 ON (E.DEPARTMENT_ID = D.DEPARTMENT_ID)
	 JOIN LOCATIONS S 
	 ON(D.LOCATION_ID = S.LOCATION_ID)
	 WHERE E.DEPARTMENT_ID = 10;
	 
	 DBMS_OUTPUT.PUT_LINE (F_NAME ||' '|| LAST_NAME ||' '||  SAL ||' '|| CITY ||' '|| DEPARTMENT_NAME );
	 
END ;

SET SERVEROUTPUT ON;

DECLARE 
    v_first_name  VARCHAR2(20);
    v_sal         NUMBER(10);
    v_new_com     NUMBER(10);
BEGIN 

    SELECT FIRST_NAME,
           SALARY,
           CASE 
               WHEN COMMISSION_PCT IS NULL THEN 0
               ELSE COMMISSION_PCT
           END
    INTO   v_first_name,
           v_sal,
           v_new_com
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID = 10
    AND ROWNUM = 1;  -- ensures single row

    DBMS_OUTPUT.PUT_LINE(
        v_first_name || ' ' || v_sal || ' ' || v_new_com
    );

END;

     DECLARE 

EMP_ID NUMBER(3) := &NUM;
FIRST_NAME VARCHAR2(20) ;
SALARY NUMBER(20);
EXP_1 NUMBER(3);

BEGIN
SELECT EMPLOYEE_ID , FIRST_NAME , SALARY , ROUND(MONTHS_BETWEEN(SYSDATE , HIRE_DATE)) AS EXP_M  INTO EMP_ID , FIRST_NAME , SALARY, EXP_1
FROM EMPLOYEES 
WHERE EMPLOYEE_ID = EMP_ID;

DBMS_OUTPUT.PUT_LINE(EMP_ID ||' '||  FIRST_NAME ||' '|| SALARY ||' '|| EXP_1);

IF EXP_1 > 10
  THEN 
   DBMS_OUTPUT.PUT_LINE('SENIOR EMP ');
   
ELSE 
   DBMS_OUTPUT.PUT_LINE('NOT SENOIR EMP ');

END IF;

END;

SET SERVEROUTPUT ON
DECLARE 

DID NUMBER := &DID_NUM;
DEPT_NAME VARCHAR2(20);
EMP_CNT NUMBER;


BEGIN 

SELECT D.DEPARTMENT_NAME , E.DEPARTMENT_ID , COUNT(EMPLOYEE_ID) AS EMP_CNT INTO  DEPT_NAME , DID , EMP_CNT 
FROM EMPLOYEES E JOIN DEPARTMENTS D
ON (E.DEPARTMENT_ID = D.DEPARTMENT_ID)
GROUP BY D.DEPARTMENT_NAME ,E.DEPARTMENT_ID 
HAVING E.DEPARTMENT_ID = DID;


DBMS_OUTPUT.PUT_LINE(DEPT_NAME || DID||' ' || EMP_CNT );


END;
