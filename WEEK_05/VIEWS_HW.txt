CREATE OR REPLACE VIEW ORDER_VIEW
AS
SELECT CUST_NAME , ORDER_NAME ,QUANTITY
FROM ORDERS2 O JOIN CUSTOMER1 C
ON(O.ORDER_ID = C.CUST_ORDER_ID);

SELECT *
FROM ORDER_VIEW;

CREATE OR REPLACE VIEW CUST_ORDER_CNT
AS 
SELECT CUST_NAME , COUNT(O.ORDER_ID) AS CNT
FROM ORDERS2 O JOIN CUSTOMER1 C
ON(O.ORDER_ID = C.CUST_ORDER_ID)
GROUP BY CUST_NAME;

SELECT *
FROM CUST_ORDER_CNT;
